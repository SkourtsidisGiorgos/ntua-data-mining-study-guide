\section{Βασικά Metrics}

\subsection{Accuracy (Ακρίβεια)}

\begin{formulabox}[Accuracy]
\begin{equation}
\text{Accuracy} = \frac{TP + TN}{TP + TN + FP + FN} = \frac{\text{Σωστές προβλέψεις}}{\text{Όλες οι προβλέψεις}}
\end{equation}
\end{formulabox}

\begin{examplebox}
\[
\text{Accuracy} = \frac{85 + 890}{1000} = \frac{975}{1000} = 0.975 = 97.5\%
\]
\end{examplebox}

\begin{warningbox}[Accuracy Paradox - Imbalanced Data]
Αν έχουμε 990 not spam και 10 spam:
\begin{itemize}
    \item Model που προβλέπει πάντα "not spam"
    \item Accuracy = $990/1000 = 99\%$... αλλά ΟΛΟΙ οι spammers περνούν!
\end{itemize}
\important{Accuracy μπορεί να είναι misleading σε imbalanced datasets!}
\end{warningbox}

\subsection{Precision (Ακρίβεια Θετικών Προβλέψεων)}

\begin{formulabox}[Precision]
\begin{equation}
\text{Precision} = \frac{TP}{TP + FP}
\end{equation}

"Από αυτά που \textbf{είπα positive}, πόσα ήταν όντως positive?"
\end{formulabox}

\begin{examplebox}
\[
\text{Precision} = \frac{85}{85 + 10} = \frac{85}{95} = 0.895 = 89.5\%
\]
\end{examplebox}

\begin{infobox}[Πότε είναι σημαντικό]
Όταν το \textbf{FP είναι costly} (π.χ. μη-spam email πάει στα spam)
\end{infobox}

\subsection{Recall / Sensitivity / True Positive Rate (TPR)}

\begin{formulabox}[Recall]
\begin{equation}
\text{Recall} = \frac{TP}{TP + FN}
\end{equation}

"Από τα \textbf{όντως positive}, πόσα βρήκα;"
\end{formulabox}

\begin{examplebox}
\[
\text{Recall} = \frac{85}{85 + 15} = \frac{85}{100} = 0.85 = 85\%
\]
\end{examplebox}

\begin{infobox}[Πότε είναι σημαντικό]
Όταν το \textbf{FN είναι costly} (π.χ. χάνουμε cancer diagnosis)
\end{infobox}

\subsection{Specificity / True Negative Rate (TNR)}

\begin{formulabox}[Specificity]
\begin{equation}
\text{Specificity} = \frac{TN}{TN + FP}
\end{equation}

"Από τα \textbf{όντως negative}, πόσα αναγνώρισα ως negative?"
\end{formulabox}

\begin{examplebox}
\[
\text{Specificity} = \frac{890}{890 + 10} = \frac{890}{900} = 0.989 = 98.9\%
\]
\end{examplebox}

\subsection{F1 Score}

Harmonic mean του Precision και Recall - \textbf{balances both metrics}.

\begin{formulabox}[F1 Score]
\begin{equation}
F_1 = \frac{2 \times \text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}} = \frac{2 \times TP}{2 \times TP + FP + FN}
\end{equation}
\end{formulabox}

\begin{examplebox}
\begin{align*}
F_1 &= \frac{2 \times 0.895 \times 0.85}{0.895 + 0.85} = \frac{1.5215}{1.745} = 0.872 = 87.2\% \\[0.5em]
\text{Εναλλακτικά}: F_1 &= \frac{2 \times 85}{2 \times 85 + 10 + 15} = \frac{170}{195} = 0.872
\end{align*}
\end{examplebox}

\begin{infobox}[Ιδιότητα F1]
F1 είναι χαμηλό αν \textbf{ένα} από Precision/Recall είναι χαμηλό!
\begin{itemize}
    \item Precision = 0.9, Recall = 0.9 $\Rightarrow$ F1 = 0.9
    \item Precision = 0.9, Recall = 0.1 $\Rightarrow$ F1 = 0.18 (χαμηλό!)
\end{itemize}
\end{infobox}

\subsection{False Positive Rate (FPR)}

\begin{formulabox}[False Positive Rate]
\begin{equation}
\text{FPR} = \frac{FP}{FP + TN} = 1 - \text{Specificity}
\end{equation}

"Ποσοστό negative που λάθος τα είπα positive"
\end{formulabox}

\section{Συχνά Λάθη}

\begin{warningbox}[Κοινά Λάθη στις Εξετάσεις]
\begin{enumerate}
    \item \textbf{Accuracy Paradox}:
    \begin{itemize}
        \item[$\times$] Υψηλό accuracy = καλό model
        \item[\checkmark] Σε imbalanced data μπορεί να είναι misleading
    \end{itemize}

    \item \textbf{Σύγχυση Precision/Recall}:
    \begin{itemize}
        \item[$\times$] Precision = $\frac{TP}{TP + FN}$
        \item[\checkmark] Precision = $\frac{TP}{TP + FP}$ (predicted positives)
        \item[\checkmark] Recall = $\frac{TP}{TP + FN}$ (actual positives)
    \end{itemize}

    \item \textbf{Specificity vs Recall}:
    \begin{itemize}
        \item Specificity = $\frac{TN}{TN + FP}$ $\rightarrow$ για τα NEGATIVE
        \item Recall = $\frac{TP}{TP + FN}$ $\rightarrow$ για τα POSITIVE
    \end{itemize}
\end{enumerate}
\end{warningbox}
