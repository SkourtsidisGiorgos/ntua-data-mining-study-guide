\section{Decision Trees}

\begin{center}
\begin{tikzpicture}
    % Entropy
    \node[draw=secondary, fill=box-blue, rounded corners=5pt, minimum width=7cm, minimum height=2.2cm] (ent) at (0,0) {};
    \node[anchor=north, font=\bfseries\color{secondary}] at (ent.north) {Entropy (Εντροπία)};
    \node[anchor=center] at (ent.center) {$H(S) = -\sum_{i=1}^{c} p_i \cdot \log_2(p_i)$};
    \node[anchor=south, font=\scriptsize] at (ent.south) {$H=0$: pure, $H=1$: max uncertainty (2 classes)};

    % Information Gain
    \node[draw=secondary, fill=box-blue, rounded corners=5pt, minimum width=7cm, minimum height=2.2cm] (ig) at (8,0) {};
    \node[anchor=north, font=\bfseries\color{secondary}] at (ig.north) {Information Gain};
    \node[anchor=center] at (ig.center) {$IG(S,A) = H(S) - \sum_{v} \frac{|S_v|}{|S|} H(S_v)$};
    \node[anchor=south, font=\scriptsize] at (ig.south) {Higher IG = Better split};

    % Gini Index
    \node[draw=success, fill=box-green, rounded corners=5pt, minimum width=7cm, minimum height=2.2cm] (gini) at (0,-3) {};
    \node[anchor=north, font=\bfseries\color{success}] at (gini.north) {Gini Index};
    \node[anchor=center] at (gini.center) {$Gini(S) = 1 - \sum_{i=1}^{c} p_i^2$};
    \node[anchor=south, font=\scriptsize] at (gini.south) {$Gini=0$: pure, $Gini=0.5$: max (2 classes)};

    % Gain Ratio
    \node[draw=success, fill=box-green, rounded corners=5pt, minimum width=7cm, minimum height=2.2cm] (gr) at (8,-3) {};
    \node[anchor=north, font=\bfseries\color{success}] at (gr.north) {Gain Ratio (C4.5)};
    \node[anchor=center] at (gr.center) {$GainRatio = \frac{IG(S,A)}{SplitInfo(S,A)}$};
    \node[anchor=south, font=\scriptsize] at (gr.south) {Corrects bias for many-valued attributes};
\end{tikzpicture}
\end{center}

\section{Association Rules}

\begin{center}
\begin{tikzpicture}
    % Support
    \node[draw=secondary, fill=box-blue, rounded corners=5pt, minimum width=6.5cm, minimum height=2.2cm] (sup) at (0,0) {};
    \node[anchor=north, font=\bfseries\color{secondary}] at (sup.north) {Support};
    \node[anchor=center] at (sup.center) {$Support(X) = \frac{|T \text{ containing } X|}{|T|}$};

    % Confidence
    \node[draw=success, fill=box-green, rounded corners=5pt, minimum width=6.5cm, minimum height=2.2cm] (conf) at (7.5,0) {};
    \node[anchor=north, font=\bfseries\color{success}] at (conf.north) {Confidence};
    \node[anchor=center] at (conf.center) {$Conf(X \to Y) = \frac{Sup(X \cup Y)}{Sup(X)}$};

    % Lift
    \node[draw=info, fill=info!20, rounded corners=5pt, minimum width=14cm, minimum height=2.2cm] (lift) at (3.75,-2.8) {};
    \node[anchor=north, font=\bfseries\color{info}] at (lift.north) {Lift};
    \node[anchor=center] at (lift.center) {$Lift(X \to Y) = \frac{Conf(X \to Y)}{Sup(Y)} = \frac{Sup(X \cup Y)}{Sup(X) \times Sup(Y)}$};
    \node[anchor=south, font=\scriptsize] at (lift.south) {$Lift=1$: independent, $>1$: positive, $<1$: negative correlation};
\end{tikzpicture}
\end{center}

\section{Quick Reference Table}

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Topic} & \textbf{Key Formula} & \textbf{Notes} \\
\midrule
Entropy & $H = -\sum p_i \log_2 p_i$ & $H=0$: pure, $H=1$: max \\
Info Gain & $IG = H(S) - \sum \frac{|S_v|}{|S|} H(S_v)$ & Higher = better \\
Gini & $G = 1 - \sum p_i^2$ & Used in CART \\
\midrule
Support & $|T \text{ with } X| / |T|$ & Frequency \\
Confidence & $Sup(X \cup Y) / Sup(X)$ & Conditional prob \\
Lift & $Conf / Sup(Y)$ & $1$=indep, $>1$=pos \\
\midrule
DTW & $d(x_i,y_j) + \min\{...\}$ & Handles time shift \\
Cosine & $(\mathbf{x} \cdot \mathbf{y}) / (\|\mathbf{x}\| \|\mathbf{y}\|)$ & Direction only \\
\midrule
Precision & $TP/(TP+FP)$ & Predicted + \\
Recall & $TP/(TP+FN)$ & Actual + \\
F1 & $2PR/(P+R)$ & Harmonic mean \\
\midrule
Bayes & $P(A|B) = P(B|A)P(A)/P(B)$ & Posterior \\
TF-IDF & $TF \times \log(N/df)$ & Word importance \\
\bottomrule
\end{tabular}
\end{center}
