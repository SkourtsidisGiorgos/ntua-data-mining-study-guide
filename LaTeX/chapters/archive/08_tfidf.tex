\chapter{Text Mining \& TF-IDF}

\section{Εισαγωγή}

\keyword{Text Mining} αναλύει και εξάγει πληροφορίες από κείμενα. \keyword{TF-IDF} είναι μέθοδος για να μετρήσουμε τη σημασία λέξεων σε documents.

\begin{definitionbox}[Bag of Words (BoW)]
\begin{itemize}
    \item Αγνοεί τη σειρά των λέξεων
    \item Μετράει μόνο frequencies
    \item Document = vector of word counts
\end{itemize}

\textbf{Παράδειγμα}:

"the cat sat on the mat" $\to$ \{the:2, cat:1, sat:1, on:1, mat:1\}
\end{definitionbox}

% =============================================================================
\section{Term Frequency (TF)}

\begin{formulabox}[Term Frequency]
\begin{equation}
TF(t, d) = \text{αριθμός εμφανίσεων του } t \text{ στο } d
\end{equation}

\textbf{Variants}:
\begin{itemize}
    \item Raw count: $tf(t,d) = f(t,d)$
    \item Boolean: $tf(t,d) = 1$ if $t \in d$, else $0$
    \item Log normalized: $tf(t,d) = 1 + \log(f(t,d))$ if $f > 0$
\end{itemize}
\end{formulabox}

\begin{examplebox}[TF Calculation]
Document: "the cat sat on the mat the cat"

\begin{center}
\begin{tabular}{cc}
\toprule
Term & TF \\
\midrule
"the" & 3 \\
"cat" & 2 \\
"sat" & 1 \\
"on" & 1 \\
"mat" & 1 \\
\bottomrule
\end{tabular}
\end{center}
\end{examplebox}

\section{Inverse Document Frequency (IDF)}

\begin{formulabox}[Inverse Document Frequency]
\begin{equation}
IDF(t) = \log\left(\frac{N}{df(t)}\right)
\end{equation}

\begin{itemize}
    \item $N$ = total number of documents
    \item $df(t)$ = number of documents containing term $t$
\end{itemize}

\textbf{Smoothed variant}: $IDF(t) = \log\left(\frac{N+1}{df(t)+1}\right) + 1$
\end{formulabox}

\section{TF-IDF}

\begin{formulabox}[TF-IDF Formula - ΑΠΟΜΝΗΜΟΝΕΥΣΕ!]
\begin{equation}
\text{TF-IDF}(t, d) = TF(t, d) \times IDF(t)
\end{equation}

\textbf{Υψηλό TF-IDF σημαίνει}:
\begin{itemize}
    \item Λέξη εμφανίζεται συχνά στο \textbf{ΣΥΓΚΕΚΡΙΜΕΝΟ} document
    \item Λέξη είναι \textbf{ΣΠΑΝΙΑ} στα υπόλοιπα documents
\end{itemize}
\end{formulabox}

\section{Document Similarity}

\begin{formulabox}[Cosine Similarity]
\begin{equation}
\text{sim}(d_1, d_2) = \frac{d_1 \cdot d_2}{\|d_1\| \times \|d_2\|} = \frac{\sum_i \text{tfidf}_1[i] \times \text{tfidf}_2[i]}{\sqrt{\sum_i \text{tfidf}_1[i]^2} \times \sqrt{\sum_i \text{tfidf}_2[i]^2}}
\end{equation}

\textbf{Range for TF-IDF vectors}: $[0, 1]$
\begin{itemize}
    \item $1$ = identical documents
    \item $0$ = completely different
\end{itemize}
\end{formulabox}

% =============================================================================
\section{Συχνά Λάθη}

\begin{warningbox}[Κοινά Λάθη στις Εξετάσεων]
\begin{enumerate}
    \item \textbf{df = 0}:
    \begin{itemize}
        \item[$\times$] $IDF = \log(N/0)$ $\to$ undefined!
        \item[\checkmark] Use smoothing: $\log(N/(df+1))$
    \end{itemize}

    \item \textbf{Log Base}:
    \begin{itemize}
        \item Δεν έχει σημασία η βάση (σταθερός συντελεστής)
        \item Συνήθως: $\log_{10}$ ή $\ln$
    \end{itemize}

    \item \textbf{IDF = 0}:
    \begin{itemize}
        \item Συμβαίνει όταν $df = N$ (λέξη σε ΟΛΑ τα docs)
        \item $IDF = \log(N/N) = \log(1) = 0$
    \end{itemize}
\end{enumerate}
\end{warningbox}
