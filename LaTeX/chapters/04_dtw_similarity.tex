\section{Μέτρα Απόστασης (Distance Metrics)}

\subsection{Euclidean Distance}

Η "ευθεία γραμμή" απόσταση στον n-διάστατο χώρο.

\begin{formulabox}[Euclidean Distance]
\begin{equation}
d(\mathbf{x}, \mathbf{y}) = \sqrt{\sum_{i=1}^{n} (x_i - y_i)^2}
\end{equation}

2D: $d = \sqrt{(x_1 - y_1)^2 + (x_2 - y_2)^2}$
\end{formulabox}

\begin{examplebox}[Υπολογισμός Euclidean]
\textbf{Points}: $\mathbf{x} = (1, 2, 3)$, $\mathbf{y} = (4, 6, 3)$

\begin{align*}
d &= \sqrt{(1-4)^2 + (2-6)^2 + (3-3)^2} \\
  &= \sqrt{9 + 16 + 0} \\
  &= \sqrt{25} = \boxed{5}
\end{align*}
\end{examplebox}

\subsection{Manhattan Distance (City Block / L1)}

Η "ταξί" απόσταση - κίνηση μόνο οριζόντια/κάθετα.

\begin{formulabox}[Manhattan Distance]
\begin{equation}
d(\mathbf{x}, \mathbf{y}) = \sum_{i=1}^{n} |x_i - y_i|
\end{equation}

2D: $d = |x_1 - y_1| + |x_2 - y_2|$
\end{formulabox}

\begin{examplebox}[Υπολογισμός Manhattan]
\textbf{Points}: $\mathbf{x} = (1, 2)$, $\mathbf{y} = (4, 6)$

\[
d = |1-4| + |2-6| = 3 + 4 = \boxed{7}
\]
\end{examplebox}

\subsection{Minkowski Distance (Generalized)}

\begin{formulabox}[Minkowski Distance]
\begin{equation}
d(\mathbf{x}, \mathbf{y}) = \left(\sum_{i=1}^{n} |x_i - y_i|^p\right)^{1/p}
\end{equation}

\begin{itemize}
    \item $p = 1$: Manhattan Distance
    \item $p = 2$: Euclidean Distance
    \item $p \to \infty$: Chebyshev Distance ($\max_i |x_i - y_i|$)
\end{itemize}
\end{formulabox}

\subsection{Cosine Similarity}

Μετράει τη γωνία μεταξύ δύο vectors (χρήσιμο για text/sparse data).

\begin{formulabox}[Cosine Similarity]
\begin{equation}
\cos(\theta) = \frac{\mathbf{x} \cdot \mathbf{y}}{\|\mathbf{x}\| \times \|\mathbf{y}\|} = \frac{\sum_i x_i y_i}{\sqrt{\sum_i x_i^2} \times \sqrt{\sum_i y_i^2}}
\end{equation}

\begin{center}
\begin{tikzpicture}
    \node[draw=success, fill=success!20, rounded corners=3pt, minimum width=2cm] at (0,0) {$\cos = 1$};
    \node[font=\small] at (0,-0.6) {Identical};

    \node[draw=warning, fill=warning!20, rounded corners=3pt, minimum width=2cm] at (4,0) {$\cos = 0$};
    \node[font=\small] at (4,-0.6) {Orthogonal};

    \node[draw=accent, fill=accent!20, rounded corners=3pt, minimum width=2cm] at (8,0) {$\cos = -1$};
    \node[font=\small] at (8,-0.6) {Opposite};
\end{tikzpicture}
\end{center}

\textbf{Cosine Distance} = $1 - \cos(\theta)$
\end{formulabox}

\begin{examplebox}[Υπολογισμός Cosine Similarity]
\textbf{Vectors}: $\mathbf{x} = (1, 2, 3)$, $\mathbf{y} = (2, 4, 6)$

\begin{align*}
\cos(\theta) &= \frac{1 \times 2 + 2 \times 4 + 3 \times 6}{\sqrt{1+4+9} \times \sqrt{4+16+36}} \\
&= \frac{2 + 8 + 18}{\sqrt{14} \times \sqrt{56}} = \frac{28}{28} = \boxed{1.0}
\end{align*}

Τα vectors έχουν ίδια κατεύθυνση ($\mathbf{y} = 2\mathbf{x}$)
\end{examplebox}

\subsection{Jaccard Similarity}

Για sets - μετράει overlap.

\begin{formulabox}[Jaccard Similarity]
\begin{equation}
J(A, B) = \frac{|A \cap B|}{|A \cup B|}
\end{equation}

\begin{itemize}
    \item Range: $[0, 1]$
    \item $J = 1$: identical sets
    \item $J = 0$: no common elements
\end{itemize}

\textbf{Jaccard Distance} = $1 - J(A, B)$
\end{formulabox}

\begin{examplebox}[Υπολογισμός Jaccard]
\textbf{Sets}: $A = \{\text{apple, banana, orange}\}$, $B = \{\text{banana, orange, grape}\}$

\begin{align*}
A \cap B &= \{\text{banana, orange}\} \quad \Rightarrow |A \cap B| = 2 \\
A \cup B &= \{\text{apple, banana, orange, grape}\} \quad \Rightarrow |A \cup B| = 4 \\[5pt]
J(A, B) &= \frac{2}{4} = \boxed{0.5}
\end{align*}
\end{examplebox}

\subsection{DTW Recurrence Relation}

\begin{formulabox}[DTW Recurrence - ΑΠΟΜΝΗΜΟΝΕΥΣΕ!]
\begin{equation}
\text{DTW}(i, j) = d(x_i, y_j) + \min \begin{cases}
\text{DTW}(i-1, j-1) & \text{(diagonal)}\\
\text{DTW}(i-1, j) & \text{(vertical)}\\
\text{DTW}(i, j-1) & \text{(horizontal)}
\end{cases}
\end{equation}

\textbf{Local cost}: $d(x_i, y_j) = |x_i - y_j|$ ή $(x_i - y_j)^2$

\textbf{Base cases}:
\begin{itemize}
    \item $\text{DTW}(0, 0) = 0$
    \item $\text{DTW}(i, 0) = \infty$ για $i > 0$
    \item $\text{DTW}(0, j) = \infty$ για $j > 0$
\end{itemize}

\textbf{Final answer}: $\text{DTW}(n, m)$
\end{formulabox}

\section{Συχνά Λάθη}

\begin{warningbox}[Κοινά Λάθη στις Εξετάσεις]
\begin{enumerate}
    \item \textbf{DTW Base Cases}:
    \begin{itemize}
        \item[$\times$] $\text{DTW}(0,j) = 0$ ή $\text{DTW}(i,0) = 0$
        \item[\checkmark] $\text{DTW}(i,0) = \text{DTW}(0,j) = \infty$ για $i,j > 0$
        \item[\checkmark] Μόνο $\text{DTW}(0,0) = 0$!
    \end{itemize}

    \item \textbf{Cost vs Distance}:
    \begin{itemize}
        \item[$\times$] $d(x,y) = x - y$ (μπορεί να είναι αρνητικό!)
        \item[\checkmark] $d(x,y) = |x - y|$ ή $(x-y)^2$
    \end{itemize}

    \item \textbf{Cosine}:
    \begin{itemize}
        \item[$\times$] Cosine similarity = distance
        \item[\checkmark] Cosine Distance $= 1 - \cos(\theta)$
    \end{itemize}

    \item \textbf{Jaccard Denominator}:
    \begin{itemize}
        \item[$\times$] $J = |A \cap B| / \max(|A|, |B|)$
        \item[\checkmark] $J = |A \cap B| / |A \cup B|$
    \end{itemize}
\end{enumerate}
\end{warningbox}
