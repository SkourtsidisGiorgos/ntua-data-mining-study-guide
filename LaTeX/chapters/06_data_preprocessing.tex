\begin{warningbox}[Common Confusion]
\textbf{Interval vs Ratio:}
\begin{itemize}
    \item Temperature in \textbf{Celsius/Fahrenheit}: Interval (0° is not "no heat")
    \item Temperature in \textbf{Kelvin}: Ratio (0K = absolute zero)
    \item \textbf{Test:} Can you say "twice as much"? If yes → Ratio
\end{itemize}
\end{warningbox}

\section{Distance Measures}

\subsection{Euclidean Distance}

\begin{formulabox}[Euclidean Distance (L2)]
\[
d_{Euclidean}(\mathbf{x}, \mathbf{y}) = \sqrt{\sum_{i=1}^{n} (x_i - y_i)^2}
\]

\textbf{Properties:}
\begin{itemize}
    \item Sensitive to scale (χρειάζεται normalization)
    \item Works for continuous attributes
\end{itemize}
\end{formulabox}

\subsection{Manhattan Distance}

\begin{formulabox}[Manhattan Distance (L1)]
\[
d_{Manhattan}(\mathbf{x}, \mathbf{y}) = \sum_{i=1}^{n} |x_i - y_i|
\]

\textbf{Properties:}
\begin{itemize}
    \item Less sensitive to outliers than Euclidean
    \item "City block" distance
\end{itemize}
\end{formulabox}

\subsection{Cosine Similarity}

\begin{formulabox}[Cosine Similarity]
\[
\cos(\theta) = \frac{\mathbf{x} \cdot \mathbf{y}}{\|\mathbf{x}\| \|\mathbf{y}\|} = \frac{\sum_i x_i y_i}{\sqrt{\sum_i x_i^2} \cdot \sqrt{\sum_i y_i^2}}
\]

\textbf{Properties:}
\begin{itemize}
    \item Range: $[-1, 1]$ (or $[0, 1]$ for non-negative vectors)
    \item Measures \textbf{direction}, ignores magnitude
    \item Common for text/document similarity
\end{itemize}
\end{formulabox}

\subsection{Jaccard Similarity}

\begin{formulabox}[Jaccard Similarity]
\[
J(A, B) = \frac{|A \cap B|}{|A \cup B|}
\]

\textbf{Properties:}
\begin{itemize}
    \item Range: $[0, 1]$ (1 = identical sets, 0 = no overlap)
    \item Best for \textbf{binary/set data}
    \item Common for document comparison, recommendation systems
\end{itemize}

\textbf{Jaccard Distance:} $d_J(A, B) = 1 - J(A, B)$
\end{formulabox}

\begin{examplebox}[Jaccard Similarity Calculation]
\textbf{Sets:} $A = \{1, 2, 3, 4\}$, $B = \{2, 3, 5, 6\}$

\begin{align*}
A \cap B &= \{2, 3\} \quad \Rightarrow |A \cap B| = 2 \\
A \cup B &= \{1, 2, 3, 4, 5, 6\} \quad \Rightarrow |A \cup B| = 6 \\
J(A, B) &= \frac{2}{6} = \frac{1}{3} \approx 0.33
\end{align*}
\end{examplebox}

\begin{examplebox}[Distance Calculation]
\textbf{Points:} $A = (1, 2)$, $B = (4, 6)$

\textbf{Euclidean:}
\[
d = \sqrt{(4-1)^2 + (6-2)^2} = \sqrt{9 + 16} = \sqrt{25} = 5
\]

\textbf{Manhattan:}
\[
d = |4-1| + |6-2| = 3 + 4 = 7
\]

\textbf{Cosine Similarity:}
\[
\cos(\theta) = \frac{1 \times 4 + 2 \times 6}{\sqrt{1^2+2^2} \cdot \sqrt{4^2+6^2}} = \frac{4 + 12}{\sqrt{5} \cdot \sqrt{52}} = \frac{16}{\sqrt{260}} \approx 0.99
\]
\end{examplebox}

\section{Data Discretization}

\subsection{Equal-Width Partitioning}

\begin{formulabox}[Equal-Width Binning]
Διαίρεση του range σε $k$ bins με \textbf{ίσο πλάτος}.

\[
\text{Width} = \frac{\max - \min}{k}
\]

Bin $i$: $[\min + (i-1) \times \text{width}, \min + i \times \text{width})$
\end{formulabox}

\subsection{Equal-Frequency (Equal-Depth) Partitioning}

\begin{formulabox}[Equal-Frequency Binning]
Διαίρεση ώστε κάθε bin να έχει \textbf{ίσο αριθμό records}.

\[
\text{Records per bin} = \frac{n}{k}
\]
\end{formulabox}

\begin{examplebox}[Discretization - 2024-25 Exam Q16 Style]
\textbf{Data:} 4, 8, 9, 15, 21, 21, 24, 25, 26, 28, 29, 34

\textbf{Equal-Width (3 bins):}
\begin{itemize}
    \item Range: 34 - 4 = 30, Width = 30/3 = 10
    \item Bin 1 [4, 14): 4, 8, 9
    \item Bin 2 [14, 24): 15, 21, 21
    \item Bin 3 [24, 34]: 24, 25, 26, 28, 29, 34
\end{itemize}

\textbf{Equal-Frequency (3 bins):}
\begin{itemize}
    \item 12 values / 3 bins = 4 per bin
    \item Bin 1: 4, 8, 9, 15
    \item Bin 2: 21, 21, 24, 25
    \item Bin 3: 26, 28, 29, 34
\end{itemize}
\end{examplebox}

\section{Common Mistakes}

\begin{warningbox}[Συχνά Λάθη στις Εξετάσεις]
\begin{enumerate}
    \item \textbf{Interval vs Ratio confusion}:
    \begin{itemize}
        \item Temperature °C: \textbf{Interval} (20°C is NOT "twice as hot" as 10°C)
        \item Height cm: \textbf{Ratio} (200cm IS twice 100cm)
    \end{itemize}

    \item \textbf{Equal-Width bin boundaries}:
    \begin{itemize}
        \item Include min, exclude max in all but last bin
        \item Last bin: $[\cdot, \max]$
    \end{itemize}

    \item \textbf{Cosine vs Euclidean}:
    \begin{itemize}
        \item Cosine: direction only (good for text)
        \item Euclidean: considers magnitude
    \end{itemize}
\end{enumerate}
\end{warningbox}
