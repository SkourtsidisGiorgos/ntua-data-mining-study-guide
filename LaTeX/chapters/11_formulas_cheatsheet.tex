    \node[anchor=center] at (vit.center) {

        Init: $\delta_1(i) = \pi_i \cdot b_i(o_1)$ \quad | \quad Backtrack using $\psi_t(j) = \arg\max_i[\delta_{t-1}(i) \cdot a_{ij}]$
    };
\end{tikzpicture}
\end{center}

\section{Bloom Filter}

\begin{center}
\begin{tikzpicture}
    \node[draw=secondary, fill=box-blue, rounded corners=5pt, minimum width=6cm, minimum height=2cm] (fp) at (0,0) {};
    \node[anchor=north, font=\bfseries] at (fp.north) {False Positive Rate};
    \node[anchor=center] at (fp.center) {$p \approx (1 - e^{-kn/m})^k$};

    \node[draw=success, fill=box-green, rounded corners=5pt, minimum width=6cm, minimum height=2cm] (opt) at (8,0) {};
    \node[anchor=north, font=\bfseries] at (opt.north) {Optimal Hash Functions};
    \node[anchor=center] at (opt.center) {$k = \frac{m}{n} \ln 2 \approx 0.693 \frac{m}{n}$};
\end{tikzpicture}
\end{center}

\begin{center}
\textbf{Key:} No false negatives! "No" = definitely not in set | "Yes" = maybe in set
\end{center}

    \node[anchor=south, font=\small] at (noir.south) {

\begin{center}
\begin{tabular}{ll}
\textbf{Equal-Width:} & Width = $\frac{\max - \min}{k}$ \\
\textbf{Equal-Frequency:} & Records per bin = $\frac{n}{k}$ \\
\textbf{Min-Max Normalization:} & $x' = \frac{x - \min}{\max - \min}$ \\
\textbf{Z-Score:} & $z = \frac{x - \mu}{\sigma}$ \\
\end{tabular}
\end{center}

\midrule

\begin{center}
