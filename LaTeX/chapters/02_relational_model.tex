\section{Βασικοί Τελεστές}

\subsection{Selection (\texorpdfstring{$\sigma$}{sigma})}

\begin{definitionbox}[Selection]
Επιλογή γραμμών που ικανοποιούν συνθήκη.

\[
\sigma_{condition}(R)
\]

\textbf{Conditions:} $=$, $\neq$, $<$, $>$, $\leq$, $\geq$, AND, OR, NOT
\end{definitionbox}

\begin{examplebox}[Selection Example]
\textbf{Table Employee:}
\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{name} & \textbf{dept} & \textbf{salary} \\
\midrule
Alice & CS & 50000 \\
Bob & Math & 45000 \\
Carol & CS & 60000 \\
\bottomrule
\end{tabular}
\end{center}

$\sigma_{dept='CS'}(Employee)$ =

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{name} & \textbf{dept} & \textbf{salary} \\
\midrule
Alice & CS & 50000 \\
Carol & CS & 60000 \\
\bottomrule
\end{tabular}
\end{center}

$\sigma_{salary > 48000 \land dept='CS'}(Employee)$ =

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{name} & \textbf{dept} & \textbf{salary} \\
\midrule
Alice & CS & 50000 \\
Carol & CS & 60000 \\
\bottomrule
\end{tabular}
\end{center}
\end{examplebox}

\subsection{Projection (\texorpdfstring{$\pi$}{pi})}

\begin{definitionbox}[Projection]
Επιλογή στηλών (και αφαίρεση duplicates).

\[
\pi_{attr_1, attr_2, ...}(R)
\]
\end{definitionbox}

\begin{examplebox}[Projection Example]
$\pi_{name, dept}(Employee)$ =

\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{name} & \textbf{dept} \\
\midrule
Alice & CS \\
Bob & Math \\
Carol & CS \\
\bottomrule
\end{tabular}
\end{center}

$\pi_{dept}(Employee)$ = \{CS, Math\} (duplicates removed!)
\end{examplebox}

\subsection{Set Operations (\texorpdfstring{$\cup$, $\cap$, $-$}{union, intersection, difference})}

\begin{warningbox}[Compatibility Requirement]
Για $\cup$, $\cap$, $-$: Οι πίνακες ΠΡΕΠΕΙ να έχουν \textbf{ίδιο schema} (ίδιος αριθμός στηλών, συμβατοί τύποι).
\end{warningbox}

\begin{examplebox}[Set Operations]
\textbf{R:} \{1, 2, 3\} \quad \textbf{S:} \{2, 3, 4\}

\begin{center}
\begin{tabular}{lll}
$R \cup S$ = \{1, 2, 3, 4\} & $R \cap S$ = \{2, 3\} & $R - S$ = \{1\} \\
\end{tabular}
\end{center}
\end{examplebox}

\section{Join Operations}

\subsection{Natural Join (\texorpdfstring{$\bowtie$}{join})}

\begin{formulabox}[Natural Join]
Ενώνει πίνακες στις \textbf{κοινές στήλες} (ίδιο όνομα \& τιμή).

\[
R \bowtie S
\]
\end{formulabox}

\begin{examplebox}[Natural Join Example]
\textbf{Student:}
\begin{tabular}{ll}
\toprule
\textbf{sid} & \textbf{name} \\
\midrule
1 & Alice \\
2 & Bob \\
\bottomrule
\end{tabular}
\quad
\textbf{Enrolls:}
\begin{tabular}{ll}
\toprule
\textbf{sid} & \textbf{course} \\
\midrule
1 & CS101 \\
1 & CS102 \\
3 & CS101 \\
\bottomrule
\end{tabular}

\vspace{0.3cm}

$Student \bowtie Enrolls$ =

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{sid} & \textbf{name} & \textbf{course} \\
\midrule
1 & Alice & CS101 \\
1 & Alice & CS102 \\
\bottomrule
\end{tabular}
\end{center}

\textit{Σημείωση: sid=3 δεν εμφανίζεται (δεν υπάρχει στο Student)}
\end{examplebox}

\subsection{Theta Join (\texorpdfstring{$\bowtie_\theta$}{theta-join})}

\begin{formulabox}[Theta Join]
Join με arbitrary condition.

\[
R \bowtie_{\theta} S = \sigma_{\theta}(R \times S)
\]
\end{formulabox}

\section{Solved Exam Problems}

\subsection{2024-25 Exam Question 6 Style}

\begin{examplebox}[Query Translation: SQL → Relational Algebra]
\textbf{Tables:}
\begin{itemize}
    \item \texttt{Student(sid, name, dept)}
    \item \texttt{Course(cid, title, credits)}
    \item \texttt{Enrolls(sid, cid, grade)}
\end{itemize}

\textbf{Query:} Find names of CS students enrolled in courses with $\geq$ 4 credits.

\textbf{SQL:}
\begin{lstlisting}[language=SQL]
SELECT DISTINCT S.name
FROM Student S, Enrolls E, Course C
WHERE S.sid = E.sid
  AND E.cid = C.cid
  AND S.dept = 'CS'
  AND C.credits >= 4;
\end{lstlisting}

\textbf{Relational Algebra - Step by Step:}

\textbf{Step 1:} Filter students from CS
\[
R_1 = \sigma_{dept='CS'}(Student)
\]

\textbf{Step 2:} Filter courses with $\geq$ 4 credits
\[
R_2 = \sigma_{credits \geq 4}(Course)
\]

\textbf{Step 3:} Join with Enrolls
\[
R_3 = R_1 \bowtie Enrolls \bowtie R_2
\]

\textbf{Step 4:} Project name
\[
Result = \pi_{name}(R_3)
\]

\textbf{Final Answer (one expression):}
\[
\pi_{name}(\sigma_{dept='CS'}(Student) \bowtie Enrolls \bowtie \sigma_{credits \geq 4}(Course))
\]
\end{examplebox}

\subsection{Division Operation}

\begin{examplebox}[Division: "For All" Queries]
\textbf{Query:} Find students enrolled in ALL courses.

\textbf{Tables:}
\begin{itemize}
    \item $Enrolls(sid, cid)$
    \item $Course(cid, title)$
\end{itemize}

\textbf{Solution:}
\[
\pi_{sid, cid}(Enrolls) \div \pi_{cid}(Course)
\]

\textbf{Division Rule:}
\[
R(A, B) \div S(B) = \text{tuples } a \text{ such that for ALL } b \in S, (a,b) \in R
\]
\end{examplebox}

\section{Common Mistakes}

\begin{warningbox}[Συχνά Λάθη στις Εξετάσεις]
\begin{enumerate}
    \item \textbf{Projection removes duplicates}:
    \begin{itemize}
        \item $\pi_{dept}(Employee)$ returns DISTINCT values!
    \end{itemize}

    \item \textbf{Natural Join matches on ALL common columns}:
    \begin{itemize}
        \item[$\times$] $R \bowtie S$ only on one column
        \item[\checkmark] Matches on ALL columns with same name
    \end{itemize}

    \item \textbf{Selection vs Projection confusion}:
    \begin{itemize}
        \item $\sigma$ = rows (horizontal slice)
        \item $\pi$ = columns (vertical slice)
    \end{itemize}

    \item \textbf{Set operations need compatible schemas}:
    \begin{itemize}
        \item Cannot do $\pi_{name}(R) \cup \pi_{dept}(S)$ if different types
    \end{itemize}
\end{enumerate}
\end{warningbox}
